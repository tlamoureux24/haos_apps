# Without an explicit ABI, Ubuntu's parser compiles against a pinned older
# feature set where sub-profiles got the parent's network rules; ABI 3.0
# (AppArmor 3.0 and later) mediates each sub-profile's network on its own.
abi <abi/3.0>,

#include <tunables/global>

# Claude Remote AppArmor profile.
#
# claude_remote: s6, nginx, ttyd, the Ingress page and the admin terminal.
#   The terminal is an interactive shell for the HA admin, so this profile
#   stays broad on executables; it denies mounts, raw sockets, ptrace and
#   writes outside the App's own data.
# claude_remote//claude: the Claude Code binary, entered on every exec of
#   /usr/local/bin/claude. No capability. Network for claude.ai and HA-MCP, writes only to
#   its home, the workspace and /tmp. Any other program it starts runs in
#   claude_shell.
# claude_remote//claude_shell: programs started by Claude Code (shell,
#   ripgrep...). No network at all, no Claude credentials, no HA-MCP URL,
#   cannot start Claude again. This is the kernel-level barrier behind the
#   managed policy that already denies Claude's shell and file tools.
